## 基础实现：手工构造词法分析器

### 1. 记号(Token)定义
```cpp
enum class TokenType {
    // 关键字
    KEYWORD_INT, KEYWORD_FLOAT, KEYWORD_IF, KEYWORD_ELSE, KEYWORD_WHILE,

    // 标识符和字面量
    IDENTIFIER, INTEGER_LITERAL, FLOAT_LITERAL, STRING_LITERAL,

    // 运算符
    PLUS, MINUS, MULTIPLY, DIVIDE,
    ASSIGN, EQUAL, NOT_EQUAL,
    LESS, GREATER, LESS_EQUAL, GREATER_EQUAL,

    // 分隔符
    LPAREN, RPAREN, LBRACE, RBRACE, SEMICOLON, COMMA,

    // 特殊
    END_OF_FILE, ERROR
};

struct Token {
    TokenType type;
    std::string lexeme;  // 记号文本
    int line;            // 行号
    int column;          // 列号

    // 辅助方法
    std::string toString() const;
    bool isOperator() const;
    bool isKeyword() const;
};
```

### 2. 状态机核心实现
```cpp
Token Lexer::scanToken() {
    skipWhitespace();

    // 记录起始位置
    size_t startLine = line_;
    size_t startColumn = column_;

    char c = advance();

    // 字符和字符串字面量
    if (c == '"') return scanString();
    if (c == '\'') return scanChar();

    // 标识符和关键字
    if (isalpha(c) || c == '_') {
        return scanIdentifier();
    }

    // 数字字面量
    if (isdigit(c)) {
        return scanNumber();
    }

    // 运算符和分隔符
    switch (c) {
        case '(': return makeToken(TokenType::LPAREN, startLine, startColumn);
        case ')': return makeToken(TokenType::RPAREN, startLine, startColumn);
        case '{': return makeToken(TokenType::LBRACE, startLine, startColumn);
        case '}': return makeToken(TokenType::RBRACE, startLine, startColumn);
        case ';': return makeToken(TokenType::SEMICOLON, startLine, startColumn);
        case ',': return makeToken(TokenType::COMMA, startLine, startColumn);
        case '+': return makeToken(TokenType::PLUS, startLine, startColumn);
        case '-': return makeToken(TokenType::MINUS, startLine, startColumn);
        case '*': return makeToken(TokenType::MULTIPLY, startLine, startColumn);
        case '/':
            // 处理注释
            if (match('/')) {
                skipLineComment();
                return getNextToken(); // 递归获取下一个记号
            }
            if (match('*')) {
                skipBlockComment();
                return getNextToken();
            }
            return makeToken(TokenType::DIVIDE, startLine, startColumn);

        case '=':
            if (match('='))
                return makeToken(TokenType::EQUAL, startLine, startColumn);
            return makeToken(TokenType::ASSIGN, startLine, startColumn);

        case '!':
            if (match('='))
                return makeToken(TokenType::NOT_EQUAL, startLine, startColumn);
            return makeErrorToken("Unexpected character", startLine, startColumn);

        case '<':
            if (match('='))
                return makeToken(TokenType::LESS_EQUAL, startLine, startColumn);
            return makeToken(TokenType::LESS, startLine, startColumn);

        case '>':
            if (match('='))
                return makeToken(TokenType::GREATER_EQUAL, startLine, startColumn);
            return makeToken(TokenType::GREATER, startLine, startColumn);

        default:
            return makeErrorToken("Unexpected character", startLine, startColumn);
    }
}
```

### 3. 标识符和关键字扫描
```cpp
Token Lexer::scanIdentifier() {
    size_t start = position_ - 1;

    // 扫描标识符字符
    while (isalnum(peek()) || peek() == '_') {
        advance();
    }

    // 提取标识符文本
    std::string text = source_.substr(start, position_ - start);

    // 检查是否为关键字
    TokenType type = keywordType(text);
    if (type != TokenType::IDENTIFIER) {
        return makeToken(type, text, line_, column_ - text.length());
    }

    return makeToken(TokenType::IDENTIFIER, text, line_, column_ - text.length());
}

TokenType Lexer::keywordType(const std::string& text) {
    static const std::map<std::string, TokenType> keywords = {
        {"int", TokenType::KEYWORD_INT},
        {"float", TokenType::KEYWORD_FLOAT},
        {"if", TokenType::KEYWORD_IF},
        {"else", TokenType::KEYWORD_ELSE},
        {"while", TokenType::KEYWORD_WHILE},
        // 更多关键字...
    };

    auto it = keywords.find(text);
    if (it != keywords.end()) {
        return it->second;
    }
    return TokenType::IDENTIFIER;
}
```

### 4. 数字字面量扫描
```cpp
Token Lexer::scanNumber() {
    size_t start = position_ - 1;
    bool isFloat = false;

    // 整数部分
    while (isdigit(peek())) {
        advance();
    }

    // 小数部分
    if (peek() == '.' && isdigit(peekNext())) {
        isFloat = true;
        advance(); // 跳过小数点

        while (isdigit(peek())) {
            advance();
        }
    }

    // 指数部分
    if ((peek() == 'e' || peek() == 'E')) {
        isFloat = true;
        advance();

        if (peek() == '+' || peek() == '-') {
            advance();
        }

        if (!isdigit(peek())) {
            return makeErrorToken("Invalid exponent", line_, column_);
        }

        while (isdigit(peek())) {
            advance();
        }
    }

    std::string text = source_.substr(start, position_ - start);

    if (isFloat) {
        return makeToken(TokenType::FLOAT_LITERAL, text, line_, column_ - text.length());
    } else {
        return makeToken(TokenType::INTEGER_LITERAL, text, line_, column_ - text.length());
    }
}
```

### 5. 字符串字面量扫描
```cpp
Token Lexer::scanString() {
    size_t start = position_ - 1;
    std::string value;

    while (peek() != '"' && !isAtEnd()) {
        // 处理转义序列
        if (peek() == '\\') {
            advance(); // 跳过反斜杠
            char escaped = advance();
            switch (escaped) {
                case 'n': value += '\n'; break;
                case 't': value += '\t'; break;
                case 'r': value += '\r'; break;
                case '\\': value += '\\'; break;
                case '"': value += '"'; break;
                case '\'': value += '\''; break;
                default:
                    return makeErrorToken("Invalid escape sequence", line_, column_);
            }
        } else {
            value += advance();
        }
    }

    if (isAtEnd()) {
        return makeErrorToken("Unterminated string", line_, column_);
    }

    advance(); // 跳过结束引号

    // 创建字符串记号（不包含引号）
    return makeToken(TokenType::STRING_LITERAL, value, line_, column_ - value.length());
}
```

## 高级实现：基于状态表的DFA

### 1. 状态表设计
```cpp
enum class State {
    START,
    IN_IDENTIFIER,
    IN_INTEGER,
    IN_FLOAT,
    IN_STRING,
    IN_CHAR,
    IN_OPERATOR,
    IN_COMMENT_LINE,
    IN_COMMENT_BLOCK,
    ACCEPT,
    ERROR
};

// 转移表
using Transition = std::function<State(char)>;
class DFALexer {
private:
    std::map<std::pair<State, char>, State> transitionTable_;
    State currentState_;

public:
    void buildTransitionTable() {
        // 标识符状态
        for (char c = 'a'; c <= 'z'; c++) {
            transitionTable_[{State::START, c}] = State::IN_IDENTIFIER;
            transitionTable_[{State::IN_IDENTIFIER, c}] = State::IN_IDENTIFIER;
        }

        // 数字状态
        for (char c = '0'; c <= '9'; c++) {
            transitionTable_[{State::START, c}] = State::IN_INTEGER;
            transitionTable_[{State::IN_INTEGER, c}] = State::IN_INTEGER;
            transitionTable_[{State::IN_FLOAT, c}] = State::IN_FLOAT;
        }

        // 更多转移规则...
    }

    Token processInput(const std::string& input) {
        currentState_ = State::START;
        size_t start = 0;

        for (size_t i = 0; i < input.length(); i++) {
            char c = input[i];

            auto it = transitionTable_.find({currentState_, c});
            if (it != transitionTable_.end()) {
                currentState_ = it->second;
            } else {
                // 处理未定义的转移
                if (currentState_ == State::IN_IDENTIFIER) {
                    return createIdentifierToken(input.substr(start, i - start));
                }
                // 其他状态的处理...
            }
        }

        return createEOFToken();
    }
};
```

### 2. 缓冲区管理
```cpp
class BufferManager {
private:
    static const size_t BUFFER_SIZE = 4096;
    std::ifstream& file_;
    char buffer1_[BUFFER_SIZE];
    char buffer2_[BUFFER_SIZE];
    char* currentBuffer_;
    size_t position_;
    bool useRefBuffer_;

public:
    BufferManager(std::ifstream& file)
        : file_(file), currentBuffer_(buffer1_),
          position_(0), useRefBuffer_(false) {

        refillBuffer(buffer1_);
        refillBuffer(buffer2_);
    }

    char getChar() {
        if (position_ >= BUFFER_SIZE) {
            // 切换缓冲区
            if (useRefBuffer_) {
                currentBuffer_ = buffer1_;
                useRefBuffer_ = false;
            } else {
                currentBuffer_ = buffer2_;
                useRefBuffer_ = true;
            }
            position_ = 0;
        }

        return currentBuffer_[position_++];
    }

    void refillBuffer(char* buffer) {
        file_.read(buffer, BUFFER_SIZE);
        if (file_.gcount() < BUFFER_SIZE) {
            // 添加EOF标记
            buffer[file_.gcount()] = '\0';
        }
    }
};
```

enum class ErrorType {

struct LexerError {

### 2. 错误恢复策略
```cpp
class ErrorHandler {
private:
    std::vector<LexerError> errors_;
    size_t maxErrors_;

public:
    void handleError(const LexerError& error) {
        errors_.push_back(error);

        if (error.severity == ErrorSeverity::FATAL) {
            throw CompilationException(error);
        }

        if (errors_.size() >= maxErrors_) {
            throw CompilationException("Too many errors");
        }
    }

    // Panic模式恢复
    void panicRecovery(Lexer& lexer) {
        // 跳过到下一个同步点（如分号、换行）
        while (!lexer.isAtEnd()) {
            Token token = lexer.peekToken();
            if (token.type == TokenType::SEMICOLON ||
                token.type == TokenType::END_OF_LINE) {
                break;
            }
            lexer.advance();
        }
    }

    // 生成错误报告
    std::string generateReport() const {
        std::ostringstream report;
        report << "Lexical Analysis Report\n";
        report << "======================\n";

        for (const auto& error : errors_) {
            report << error.format() << "\n";
        }

        report << "\nTotal: " << errors_.size() << " error(s)\n";
        return report.str();
    }
};
```

## 性能优化技术

### 1. 关键字查找优化
```cpp
class KeywordRecognizer {
private:
    // 使用Trie树优化关键字识别
    struct TrieNode {
        std::map<char, TrieNode*> children;
        TokenType* tokenType;
    };

    TrieNode* root_;

public:
    KeywordRecognizer() {
        root_ = new TrieNode();
        insertKeyword("int", TokenType::KEYWORD_INT);
        insertKeyword("float", TokenType::KEYWORD_FLOAT);
        insertKeyword("if", TokenType::KEYWORD_IF);
        // 更多关键字...
    }

    std::pair<TokenType, size_t> matchKeyword(const std::string& text) {
        TrieNode* current = root_;
        size_t length = 0;

        for (char c : text) {
            if (current->children.find(c) == current->children.end()) {
                break;
            }
            current = current->children[c];
            length++;

            if (current->tokenType != nullptr) {
                return {*current->tokenType, length};
            }
        }

        return {TokenType::IDENTIFIER, 0};
    }
};
```

### 2. 并行处理
```cpp
class ParallelLexer {
private:
    std::vector<std::unique_ptr<Lexer>> workers_;
    std::queue<std::string> inputQueue_;
    std::queue<Token> outputQueue_;
    std::mutex queueMutex_;

public:
    void processParallel(const std::string& source) {
        // 将源代码分割为多个块
        auto chunks = splitSource(source);

        // 并行处理各个块
        std::vector<std::future<std::vector<Token>>> futures;

        for (const auto& chunk : chunks) {
            futures.push_back(std::async(std::launch::async, [chunk]() {
                Lexer lexer(chunk);
                return lexer.tokenizeAll();
            }));
        }

        // 收集结果
        for (auto& future : futures) {
            auto tokens = future.get();
            for (const auto& token : tokens) {
                outputQueue_.push(token);
            }
        }
    }
};
```
